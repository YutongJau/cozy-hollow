function blank(w, h) { const c = document.createElement("canvas"); c.width = w; c.height = h; return c; }
function px(g, x, y, w, h, col) { g.fillStyle = col; g.fillRect(x, y, w, h); }
function hash(x, y) { let h = (x * 73856093) ^ (y * 19349663); h = Math.imul(h ^ (h >>> 13), 1274126177); return (h ^ (h >>> 16)) >>> 0; }
function rng(seed) { let s = seed >>> 0 || 1; return () => { s = Math.imul(s, 1103515245) + 12345 >>> 0; return s / 4294967296; }; }
function hex(h) { const n = parseInt(h.slice(1), 16); return [(n >> 16) & 255, (n >> 8) & 255, n & 255]; }
function rgb(r, g, b) { return "rgb(" + Math.round(r) + "," + Math.round(g) + "," + Math.round(b) + ")"; }
function shade(h, f) { const [r, g, b] = hex(h); const c = (v) => Math.max(0, Math.min(255, v * f)); return rgb(c(r), c(g), c(b)); }
function mix(h1, h2, t) { const a = hex(h1), b = hex(h2); return rgb(a[0] + (b[0] - a[0]) * t, a[1] + (b[1] - a[1]) * t, a[2] + (b[2] - a[2]) * t); }
function cached(key, w, h, draw) {
  let c = cache.get(key);
  if (c) return c;
  c = blank(w, h); draw(c.getContext("2d"), c); cache.set(key, c); return c;
}
/* dark 1px outline around every opaque pixel of a canvas (the Cattle Country look) */
function outline(c, color, inner) {
  const g = c.getContext("2d"), w = c.width, h = c.height;
  const img = g.getImageData(0, 0, w, h), d = img.data;
  const solid = new Uint8Array(w * h);
  for (let i = 0; i < w * h; i++) solid[i] = d[i * 4 + 3] > 40 ? 1 : 0;
  const [r, gg, b] = hex(color);
  for (let y = 0; y < h; y++) for (let x = 0; x < w; x++) {
    const i = y * w + x;
    if (solid[i]) continue;
    const n = (x > 0 && solid[i - 1]) || (x < w - 1 && solid[i + 1]) || (y > 0 && solid[i - w]) || (y < h - 1 && solid[i + w]);
    if (n) { d[i * 4] = r; d[i * 4 + 1] = gg; d[i * 4 + 2] = b; d[i * 4 + 3] = 255; }
  }
  g.putImageData(img, 0, 0);
  return c;
}
function mirrored(key, src) {
  return cached(key, src.width, src.height, (g) => { g.translate(src.width, 0); g.scale(-1, 1); g.drawImage(src, 0, 0); });
}

function noise2(x, y) {
  /* smooth value noise on a 4-tile lattice, 0..1 */
  const x0 = Math.floor(x / 4), y0 = Math.floor(y / 4), fx = (x % 4) / 4, fy = (y % 4) / 4;
  const v = (a, b) => (hash(a * 17 + 3, b * 29 + 7) % 1000) / 1000;
  const sx = fx * fx * (3 - 2 * fx), sy = fy * fy * (3 - 2 * fy);
  const top = v(x0, y0) * (1 - sx) + v(x0 + 1, y0) * sx, bot = v(x0, y0 + 1) * (1 - sx) + v(x0 + 1, y0 + 1) * sx;
  return top * (1 - sy) + bot * sy;
}

function text3x5(g, x, y, str, col, sc) {
  sc = sc || 1;
  for (const ch of str.toUpperCase()) {
    const bits = FONT[ch] || FONT[" "];
    for (let i = 0; i < 15; i++) if (bits[i] === "1") px(g, x + (i % 3) * sc, y + Math.floor(i / 3) * sc, sc, sc, col);
    x += 4 * sc;
  }
}

// ------------------------------------------------------------ trees
function puff(g, cx, cy, r, pal, seed) {
  /* one round leaf cluster: dark rim, mid body, light cap on the upper left, a few sparkles */
  const hx = cx - r * 0.3, hy = cy - r * 0.35;
  for (let y = Math.floor(cy - r); y <= cy + r; y++) for (let x = Math.floor(cx - r); x <= cx + r; x++) {
    const d2 = (x - cx) * (x - cx) + (y - cy) * (y - cy);
    if (d2 > r * r) continue;
    const hd = Math.sqrt((x - hx) * (x - hx) + (y - hy) * (y - hy));
    let col = d2 > (r - 1.6) * (r - 1.6) ? pal.D : (hd < r * 0.42 ? pal.L : (hd < r * 0.78 ? pal.M : pal.D));
    if (col === pal.L && (hash(x * 3 + seed, y * 5 + seed) % 7) === 0) col = pal.X;
    if (col === pal.M && (hash(x + seed * 7, y * 3) % 11) === 0) col = pal.L;
    px(g, x, y, 1, 1, col);
  }
}

  const skin = "#f4c9a0", skinS = "#d9a678", hair = "#241b17", hairL = "#403129", eye = "#2a1f1a";
  const boy = kind !== "girl";
  const hat = boy ? "#8a5a3a" : "#d9b86a", hatD = boy ? "#6a4228" : "#b3934c", hatL = boy ? "#a67248" : "#eacf88", band = boy ? "#4a2e1c" : "#c94a5a";
  const shirt = shirtHex, shirtD = shade(shirtHex, 0.72), shirtL = shade(shirtHex, 1.18);
  const denim = "#4c6d9c", denimD = "#37507a", boot = "#4a2e1a", bootL = "#6a4630", apron = "#f6e7c8", apronD = "#d9c7a6";
  // smooth 8-frame cycle: legs and arms swing on a sine, body bounces twice per cycle
  const t = (frame % 8) / 8, sn = Math.sin(t * Math.PI * 2);
  const liftL = Math.round(Math.max(0, sn) * 3), liftR = Math.round(Math.max(0, -sn) * 3);
  const swing = Math.round(sn * 2), bob = frame % 8 === 0 ? 0 : Math.round(Math.abs(sn) * 1.6);
  const f = Math.round(sn * 4);                     // side-view stride
  g.translate(0, 0);
  // --- legs (drawn first)
  if (boy) {
    if (!side) {
      px(g, 10, 33 - liftL, 5, 10, denim); px(g, 10, 33 - liftL, 1, 10, denimD); px(g, 10, 42 - liftL, 5, 4, boot); px(g, 10, 42 - liftL, 5, 1, bootL);
      px(g, 17, 33 - liftR, 5, 10, denim); px(g, 21, 33 - liftR, 1, 10, denimD); px(g, 17, 42 - liftR, 5, 4, boot); px(g, 17, 42 - liftR, 5, 1, bootL);
    } else {
      px(g, 12 - f, 33, 5, 10 - Math.abs(f) / 2, denimD); px(g, 12 - f, 42 - Math.floor(Math.abs(f) / 2), 5, 4, boot);
      px(g, 15 + f, 33, 5, 10, denim); px(g, 15 + f, 42, 6, 4, boot); px(g, 15 + f, 42, 6, 1, bootL);
    }
  } else {
    if (!side) {
      px(g, 11, 38 - liftL, 4, 5, skin); px(g, 11, 42 - liftL, 4, 4, boot); px(g, 11, 42 - liftL, 4, 1, bootL);
      px(g, 17, 38 - liftR, 4, 5, skin); px(g, 17, 42 - liftR, 4, 4, boot); px(g, 17, 42 - liftR, 4, 1, bootL);
    } else {
      px(g, 13 - f, 38, 4, 5, skinS); px(g, 13 - f, 42 - Math.floor(Math.abs(f) / 3), 4, 4, boot); px(g, 15 + f, 38, 4, 5, skin); px(g, 15 + f, 42, 5, 4, boot); px(g, 15 + f, 42, 5, 1, bootL);
    }
  }
  g.translate(0, -bob);                            // everything above the legs bounces
  // --- long hair behind body (girl)
  if (!boy && !side) { px(g, 5, 8, 3, 24, hair); px(g, 24, 8, 3, 24, hair); px(g, 6, 12, 1, 12, hairL); }
  if (!boy && back) { px(g, 7, 14, 18, 20, hair); px(g, 9, 16, 2, 14, hairL); }
  if (!boy && side) { px(g, 6, 8, 6, 24, hair); px(g, 7, 12, 1, 12, hairL); }
  // --- body
  if (boy) {
    px(g, 9, 20, 14, 14, shirt); px(g, 21, 20, 2, 14, shirtD); px(g, 9, 20, 2, 3, shirtL);
    if (!side) { px(g, 12, 24, 8, 10, denim); px(g, 12, 20, 2, 5, denim); px(g, 18, 20, 2, 5, denim); px(g, 13, 26, 6, 5, denimD); px(g, 14, 27, 4, 3, denim); px(g, 12, 24, 8, 1, "#6a8fbf"); }
    else { px(g, 12, 24, 9, 10, denim); px(g, 15, 20, 2, 5, denim); px(g, 12, 24, 9, 1, "#6a8fbf"); }
    if (back) { px(g, 12, 20, 2, 6, denim); px(g, 18, 20, 2, 6, denim); px(g, 12, 25, 8, 9, denim); }
  } else {
    px(g, 9, 20, 14, 11, shirt); px(g, 21, 20, 2, 11, shirtD); px(g, 9, 20, 2, 3, shirtL);
    for (let y = 30; y < 40; y++) { const w = 14 + Math.floor((y - 30) * 0.7); const x = 16 - Math.floor(w / 2); px(g, x, y, w, 1, y >= 38 ? shirtD : shirt); px(g, x + w - 2, y, 2, 1, shirtD); }
    if (!back) { px(g, 12, 25, 8, 12, apron); px(g, 12, 25, 8, 1, apronD); px(g, 13, 35, 6, 2, apronD); px(g, 14, 21, 4, 4, apron); }
  }
  // --- arms (raised overhead when carrying something to plant)
  if (armsUp) {
    if (!side) { px(g, 5, 10, 3, 12, shirt); px(g, 5, 7, 3, 3, skin); px(g, 24, 10, 3, 12, shirtD); px(g, 24, 7, 3, 3, skin); }
    else { px(g, 18, 9, 4, 13, shirtD); px(g, 18, 6, 4, 3, skin); }
  } else if (!side) {
    px(g, 6, 21 + swing, 3, 10, shirt); px(g, 6, 30 + swing, 3, 3, skin);
    px(g, 23, 21 - swing, 3, 10, shirtD); px(g, 23, 30 - swing, 3, 3, skin);
  } else { px(g, 17 + swing, 22, 4, 9, shirtD); px(g, 17 + swing, 30, 4, 3, skin); }
  // --- head
  const hx = side ? 8 : 7;
  px(g, hx + 1, 5, 16, 17, skin); px(g, hx, 7, 18, 13, skin); px(g, hx + 16, 7, 2, 13, skinS); px(g, hx + 2, 21, 14, 1, skinS);
  if (back) { px(g, hx, 5, 18, 17, hair); px(g, hx + 2, 8, 2, 8, hairL); }
  else if (side) {
    // hair wraps the back of the head and shows under the brim; face at the front (right)
    px(g, hx, 5, 12, 6, hair); px(g, hx, 9, 14, 4, hair); px(g, hx, 12, 5, 10, hair); px(g, hx + 5, 12, 2, 2, hair); px(g, hx + 9, 12, 2, 1, hair); px(g, hx + 12, 12, 2, 1, hair);
    px(g, hx + 1, 13, 1, 6, hairL);
  } else {
    px(g, hx, 5, 18, 6, hair); px(g, hx, 9, 18, 4, hair); px(g, hx, 12, 3, 6, hair); px(g, hx + 15, 12, 3, 6, hair);
    px(g, hx + 3, 12, 2, 2, hair); px(g, hx + 7, 12, 3, 2, hair); px(g, hx + 12, 12, 2, 1, hair); px(g, hx + 1, 6, 2, 2, hairL);
  }
  if (!back) {
    if (side) { px(g, hx + 12, 14, 3, 4, eye); px(g, hx + 12, 14, 1, 1, "#fff"); px(g, hx + 11, 19, 2, 1, "#f0a0a0"); px(g, hx + 16, 16, 1, 2, skinS); px(g, hx + 14, 20, 2, 1, "#c06a5a"); }
    else { px(g, hx + 4, 14, 3, 4, eye); px(g, hx + 4, 14, 1, 1, "#fff"); px(g, hx + 11, 14, 3, 4, eye); px(g, hx + 11, 14, 1, 1, "#fff"); px(g, hx + 2, 19, 2, 1, "#f0a0a0"); px(g, hx + 14, 19, 2, 1, "#f0a0a0"); px(g, hx + 8, 20, 2, 1, "#c06a5a"); }
  }
  if (hatKind === "none") {
    // full head of hair instead of a hat: rounded top, boy short and tousled, girl with a bow
    px(g, hx + 1, 2, 16, 4, hair); px(g, hx + 3, 1, 12, 1, hair); px(g, hx, 4, 18, 3, hair); px(g, hx + 2, 3, 3, 1, hairL); px(g, hx + 5, 2, 2, 1, hairL);
    if (boy) { px(g, hx + 4, 0, 3, 2, hair); px(g, hx + 9, 0, 2, 2, hair); px(g, hx + 13, 1, 3, 1, hair); }
    else if (!back) { px(g, hx + 13, 2, 5, 3, "#e04a5a"); px(g, hx + 14, 1, 3, 5, "#e04a5a"); px(g, hx + 15, 3, 1, 1, "#ffb0b8"); px(g, hx + 13, 5, 1, 4, "#e04a5a"); }
  } else {
    // --- hat (cowboy): brim, crown, band
    const bx = side ? hx - 3 : hx - 5, bw = side ? 24 : 28;
    px(g, bx, 7, bw, 3, hat); px(g, bx + 1, 6, bw - 2, 1, hat); px(g, bx, 9, bw, 1, hatD); px(g, bx + 2, 6, 5, 1, hatL);
    px(g, hx + 2, 0, 14, 8, hat); px(g, hx + 3, -1, 12, 1, hat); px(g, hx + 6, 0, 6, 1, hatD); px(g, hx + 2, 0, 2, 7, hatL); px(g, hx + 14, 0, 2, 7, hatD);
    px(g, hx + 2, 5, 14, 2, band);
    if (!boy && !back) { px(g, hx + 14, 3, 4, 3, "#e04a5a"); px(g, hx + 15, 2, 2, 5, "#e04a5a"); px(g, hx + 15, 4, 1, 1, "#ffb0b8"); }
  }
  g.setTransform(1, 0, 0, 1, 0, 0);
  outline(c, OUT);
  const done = facing === "left" ? mirrored(key, c) : c;
  cache.set(key, done);
  return done;
}

return { A, cache, blank, px, hash, shade, mix, outline, mirrored, cached, COLORS, ITEM_HEX, SEASON, TREE_PAL, WOOD, OUT, GROUND, SOIL_EDGE,

  cropSprite, forageSprite, iconCanvas, toolSprite, farmerSprite };
